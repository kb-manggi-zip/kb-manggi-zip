```python
# app/tools/regions.py (신규)
def classify_region(region_name: str, rules) -> dict:
    """동/구/시 문자열 → {is_metro, is_regulated}"""
    is_metro = any(m in region_name for m in rules.metro)
    is_regulated = ("서울" in region_name) or \
                   any(g in region_name for g in rules.regulated.gyeonggi)
    return {"is_metro": is_metro, "is_regulated": is_regulated}
```

- ⚠️ 데모 동네(마포 합정동·은평 녹번동·도봉 창동, 성북 길음동, 노원 상계동, 중랑 면목동)는 **전부 서울 = 규제지역**
- 비교표 assumptions에 자동 노출: `"○○동은 규제지역이라 LTV 40%가 적용돼요"`

```python
def calc_buy(lease, finance, region, rules) -> BranchResult:
    r = classify_region(region, rules.regions)
    capital = finance.own_capital + lease.deposit       # 돌려받는 보증금 합산

    # ① 정책대출(디딤돌) 자격·한도 — kb_purchase 캡에서 제외됨
    policy = calc_didimdol_eligibility(finance, r, rules.policy_loans)
    #   → {eligible: bool, limit: int, rate: float, reason: str}

    # ② 은행 주담대(KB) 한도
    ltv = rules.lending.ltv.no_house.regulated if r["is_regulated"] \
          else rules.lending.ltv.no_house.non_regulated
    if finance.first_home:                              # 생애최초 플래그(선택 구현)
        ltv = rules.lending.ltv.first_home.regulated if r["is_regulated"] \
              else rules.lending.ltv.first_home.non_regulated

    # DSR 한도 (스트레스 반영) — 정책+은행 합산 원리금 기준
    dsr_rate = effective_dsr_rate(r, rules.lending.dsr, rules.lending.rates.kb_mortgage_default)
    monthly_capacity = finance.annual_income * rules.lending.dsr.cap / 12 - finance.existing_debt_monthly
    dsr_limit = pv_annuity(monthly_capacity, dsr_rate, rules.lending.loan_term_years)

    # ③ 캡 적용 — 여기가 핵심
    bank_cap = rules.lending.mortgage_cap.kb_purchase        # KB 자체 3억
    if r["is_metro"]:
        bank_cap = min(bank_cap, rules.lending.mortgage_cap.gov_metro)  # 정부 6억

    # 집값 상한: 자본 + 정책대출 + 은행대출, 단 LTV는 집값 기준이므로 역산 필요
    max_price = solve_max_price(
        capital=capital, ltv=ltv, dsr_limit=dsr_limit,
        policy_limit=policy.limit if policy.eligible else 0,
        bank_cap=bank_cap,
    )
    # solve_max_price: price를 미지수로 두고
    #   loan_needed = price - capital
    #   loan_allowed = min(price*ltv, dsr_limit, policy_limit + bank_cap)
    #   loan_needed <= loan_allowed 를 만족하는 최대 price (이분탐색 또는 폐형해)

    # ④ 월 상환 — 표시용은 실제 금리(스트레스 아님!)
    policy_amt = min(policy.limit, needed) if policy.eligible else 0
    bank_amt = needed - policy_amt
    monthly = annuity_payment(policy_amt, policy.rate, term) \
            + annuity_payment(bank_amt, rules.lending.rates.kb_mortgage_default, term)

    return BranchResult(
        branch="매매",
        headline=monthly,
        depositOrPrice=max_price,
        loanAmount=needed,
        oneTimeCost=acquisition_cost(max_price, finance.first_home),
        guaranteeMonthly=0,
        basis=[
            {"label": f"LTV {int(ltv*100)}%", "detail": f"{region} 규제지역 기준" if r["is_regulated"] else "비규제지역"},
            {"label": "KB 한도 3억", "detail": "KB국민은행 자체 정책(2026.7.10~)"},
            {"label": "스트레스 DSR", "detail": f"가산 {dsr_rate:.2%} 적용"},
            {"label": "디딤돌 혼합", "detail": f"{policy.rate:.2%}"} if policy.eligible else None,
        ],
        ...
    )
```

### 스트레스 DSR 공식 (정식)

```python
def effective_dsr_rate(region, dsr_rules, base_rate) -> float:
    """DSR '산정용' 금리. 월상환 표시에는 절대 쓰지 않는다."""
    if region["is_metro"] and region["is_regulated"]:
        stress = dsr_rules.stress_rate.metro_regulated      # 3.0%
    elif region["is_regulated"]:
        stress = dsr_rules.stress_rate.local_regulated      # 1.5%
    else:
        stress = dsr_rules.stress_rate.others               # 1.5%
    applied = stress * dsr_rules.base_ratio \
            * dsr_rules.loan_type_ratio[dsr_rules.default_loan_type]
    return base_rate + applied
```

> ⚠️ **금리 용도 분리 필수:** 한도 산정 = `base + stress`, 월 상환액 표시 = `base`만. 지금 코드는 stressRate 하나로 둘 다 쓰고 있어 월 부담이 과대 표시됨.

## 2-4. `calc_guarantee` — 3중 테이블 조회

```python
def calc_guarantee(deposit, rules, house_type=None, debt_ratio=None) -> dict:
    ht = house_type or rules.guarantee.default_house_type
    dr = debt_ratio or rules.guarantee.default_debt_ratio
    band = ("under_90m" if deposit <= 90_000_000
            else "m90_to_200m" if deposit <= 200_000_000
            else "over_200m")
    rate = rules.guarantee.hug_fee_rate[band][ht][dr]
    total = deposit * rate * (rules.guarantee.contract_days / 365)
    return {
        "annual_fee": int(deposit * rate),
        "monthly_equiv": int(total / 24),      # 2년 계약 → 월 환산
        "rate": rate,
        "basis": "HUG 공시 요율 (보증금액×보증료율×계약일수/365)",
        "assumption": f"{ht} · 부채비율 {dr} 가정",
    }
```

# PART 5. 발표 포인트 (이 반영으로 생긴 것)

> ### "저희 계산기는 규제를 세 겹으로 반영합니다."
> 1. **정부 규제** — 규제지역 LTV 40%, 수도권 주담대 6억 한도
> 2. **KB 자체 여신 정책** — 2026년 7월 10일부터 시행된 주택구입 주담대 3억 한도
> 3. **스트레스 DSR** — 수도권 규제지역 가산 3.0%를 한도 산정에만 적용(월 상환액에는 미적용)
>
> "특히 두 번째는, KB 앱 안의 서비스이기 때문에 반영할 수 있는 부분입니다.
> 정부 기준만 쓰면 고객에게 실제보다 큰 한도를 보여주게 됩니다."

이 문장이 **"왜 은행이 만들어야 하는가"에 대한 가장 강력한 증거**다. 외부 플랫폼은 KB 내부 여신 정책을 반영할 수 없다.
